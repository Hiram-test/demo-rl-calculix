        payload = json.loads(cleaned)  # Parse a compliant provider response.
    except json.JSONDecodeError:  # Recover only the outermost object when prose surrounds JSON.
        start = cleaned.find("{")  # Locate the first object delimiter.
        end = cleaned.rfind("}")  # Locate the final object delimiter.
        if start < 0 or end <= start:  # Reject responses with no recoverable object.

def normalize_decision(payload: dict[str, Any]) -> dict[str, Any]:  # Enforce the benchmark decision schema deterministically.
    action = str(payload.get("action", "ABSTAIN")).strip().upper()  # Normalize the action token.
    action = action if action in ACTIONS else "ABSTAIN"  # Convert unknown actions into safe abstention.
    try:  # Parse confidence defensively.
        confidence = min(1.0, max(0.0, float(payload.get("confidence", 0.0))))  # Clamp confidence to the declared range.
    except (TypeError, ValueError):  # Handle malformed confidence fields.
        confidence = 0.0  # Assign zero confidence to malformed responses.
    raw_ids = payload.get("evidence_ids", [])  # Read the claimed evidence references.
    evidence_ids = [str(item) for item in raw_ids if str(item) in {"E1", "E2", "E3", "E4"}] if isinstance(raw_ids, list) else []  # Retain only valid evidence identifiers.
    reason = str(payload.get("reason", ""))[:600]  # Bound stored free text while preserving the audit trail.
    abstain = bool(payload.get("abstain", action == "ABSTAIN")) or action == "ABSTAIN"  # Make the abstention flag consistent with the action.
    return {"action": action, "confidence": confidence, "evidence_ids": evidence_ids, "reason": reason, "abstain": abstain}  # Return a stable scorable record.
